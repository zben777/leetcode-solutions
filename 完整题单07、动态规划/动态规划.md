```
class Solution {
public:
    int fib(int N) {
        if (N <= 1) return N;
        vector<int> dp(N + 1);
        dp[0] = 0;
        dp[1] = 1;
        for (int i = 2; i <= N; i++) {
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[N];
    }
};
```

### 题目来源：LeetCode-746-使用最小花费爬楼梯

题目描述：  
 ![在这里插入图片描述](images/129574245_4.png)

题目思路：

> 五部曲，然后这个主要是动态规划的状态转移方程式：还是按照题意进行最顺畅

题目代码：

```
class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
        int n=cost.size();
        if(n<=1)return 0;

        vector<int>dp (n+1);

        //从下标0开始
        dp[0]=0;
        dp[1]=cost[0];
        for(int i=2;i<=n;i++){
            dp[i]=min(dp[i-1]+cost[i-1],dp[i-2]+cost[i-2]);
        }
        int minn=dp[n];

        //从下标1开始
        dp[0]=0;
        dp[1]=0;
        for(int i=2;i<=n;i++){
            dp[i]=min(dp[i-1]+cost[i-1],dp[i-2]+cost[i-2]);
        }
        dp[n]=min(minn,dp[n]);
        return dp[n];
    }
};
```

### 题目来源：LeetCode-198. 打家劫舍

> 五部曲

题目代码：

```
class Solution {
public:
    int rob(vector<int>& nums) {

        // 这个是不能连着，五部曲

        int n = nums.size();
        if(n <= 1)return nums[0];
        std::vector<int>dp(n, 0);
        dp[0] = nums[0];
        dp[1] = max(nums[0], nums[1]);

        // 每次循环到pos的时候就当是最后一个位置pos
        for(int i = 2; i < n; i++){
            dp[i] = dp[i - 1]; //此次不搞
            // 搞此次
            if(dp[i - 2] + nums[i] > dp[i]){
                dp[i] = dp[i - 2] + nums[i];
            }
        }
        return dp[n - 1];
    }
};
```

### 题目来源：LeetCode-279. 完全平方数

```
class Solution {
public:
    int numSquares(int n) {

        // 先把数据存放在vector
        // 因为最大n = 10000
        std::vector<int>ans;
        for(int i = 1; i <= 100; i++){
            ans.push_back(i * i);
        }
        std::vector<int>dp(n + 10, 0x3f3f3f3f);
        dp[0] = 0;
        dp[1] = 1;
        dp[2] = 2;
        dp[3] = 3;
        dp[4] = 1;
        // 转移公式, 两层循环
        for(int i = 5; i <= n; i++){
            // 选择其中的一个，然后+1；
            for(int j = 0; j < ans.size() && ans[j] <= i; j++){
                int now = ans[j];
                dp[i] = min(dp[i], dp[i - now] + 1);
            }
        }
        return dp[n];
    }
};
```

```
class Solution {
public:
    int maxSubArray(vector<int>& nums) {

        //五步骤：
        // 对于每个点的时候 都会面临选不选；
        // 选即加入，不选即重新一个序列的开始。
        // 选即更新，不选即
        int n = nums.size();
        int *dp = (int*)malloc(sizeof(int)*(n+1));

        int maxx = -0x3f3f3f3f;

        }

    }
```

```
class Solution {
public:
    int integerBreak(int n) {

        for(int i=3;i<=n;i++){
            dp[i]=dp[i-1];

            }
        }


        return dp[n];


    }
```

```
        int n = s.size();
        
        // 用集合快速查找
        unordered_set<string> wordSet(wordDict.begin(), wordDict.end());
        
        // dp[i] 表示：s[0...i-1] 能不能拆分成字典里的单词
        vector<bool> dp(n + 1, false);
        
        // 初始状态：0 个字符，能拆分（空串）
        dp[0] = true;

        // 开始 DP
        for (int i = 1; i <= n; i++) {       // 枚举结尾位置 i
            for (int j = 0; j < i; j++) {    // 枚举分割点 j
                // 如果：前 j 个能拆分 + 从 j 到 i 的串在字典里
                if (dp[j] && wordSet.count(s.substr(j, i - j))) {
                    dp[i] = true;
                    break; // 找到一个就行，不用继续找
                }
            }
        }

        // 答案：前 n 个字符能否拆分
        return dp[n];
    }
};
```

```
                  tails[l] = num;
              }
          }

      }
```

### 题目来源：LeetCode-152. 乘积最大子数组

题目描述：  
 ![在这里插入图片描述](images/129574245_17.png)

题目思路：

> 要注意 负负得正的事情，  
>  还要额外记录一个min\_dp

题目代码：

```
class Solution {
public:
    int maxProduct(vector<int>& nums) {

        // 这个相对于上面那个多了个连续；
        // 动态规划策略
        // dp[i] 表示：以第 i 个数字结尾的最大乘积累
        // 这个表示 第 i 个 是 必须 选择的
        int n = nums.size();
        if(n <= 1)return nums[0];
        
        // 因为乘积存在负负得正，所以必须同时维护最大值和最小值
        vector<int> max_dp(n);  // 以 i 结尾的最大乘积
        vector<int> min_dp(n);  // 以 i 结尾的最小乘积

        // 每个最大乘积累 先是自己本身
        for(int i = 0; i < n; i++) {
            max_dp[i] = nums[i];
            min_dp[i] = nums[i];
        }

        int res = nums[0]; // 答案初始化为第一个数

        // 你指定要用的 for 循环，我完完整整保留！
        for (int i = 1; i < n; i++) {
            // 之前是找前面的所有，现在仅仅需要找一个就行
            // 目前的 i 位置是 必须选的， 因为是以 i 为结尾的
            // 当前是正数：大×更大，小×更小
            if (nums[i] >= 0) {
                max_dp[i] = max(nums[i], max_dp[i-1] * nums[i]);
                min_dp[i] = min(nums[i], min_dp[i-1] * nums[i]);
            }
            // 当前是负数：大×小 反而变最大！
            else {
                max_dp[i] = max(nums[i], min_dp[i-1] * nums[i]);
                min_dp[i] = min(nums[i], max_dp[i-1] * nums[i]);
            }
            res = max(res, max_dp[i]);
        }

        return res;
    }
};
```

```
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        // 两个子集合 和 相等
        // 直接转换 为 挑出部分数组 能否 == sum / 2；

        int n = nums.size();
        int sum = 0;
        for (int x : nums) sum += x;

        // 和是奇数，直接不能分
        if (sum % 2 != 0) return false;
        int target = sum / 2;

        // ==================== 二维 DP ====================
        // dp[i][j] 含义：
        // 前 i 个数字，能否凑出 和为 j
        // true = 能，false = 不能
        vector<vector<bool>> dp(n + 1, vector<bool>(target + 1, false));

        // 初始状态：前 i 个数字，凑 0，都能凑出来（不选）
        for (int i = 0; i <= n; i++) {
            dp[i][0] = true;
        }

        // 开始填二维表
        for (int i = 1; i <= n; i++) {      // 遍历第 i 个数字
            int num = nums[i - 1];          // 当前数字
            for (int j = 1; j <= target; j++) {  // 遍历和 j
                // 不选当前数字 → 直接继承上一行
                dp[i][j] = dp[i - 1][j];

                // 能选的话，看看选了能不能行
                if (j >= num) {
                    dp[i][j] = dp[i][j] || dp[i - 1][j - num];
                }
            }
        }

        // 答案：前n个数字，能否凑出target
        return dp[n][target];
    }
};
```

```
            l--;
            r++;
        }

            expan(i, i, s);
            // 偶数个
            expan(i, i + 1, s);
        }
        return s.substr(start, maxLen);
    }
```

```
#include <vector>

class Solution {
public:
    string longestPalindrome(string s) {
        int n = s.size();
        if (n < 2) return s;

        int maxLen = 1;
        int begin = 0;
        // 第一步先定义
        // dp[i][j] 表示 s[i..j] 是否是回文串
        vector<vector<int>> dp(n, vector<int>(n));
        // 初始化：所有长度为 1 的子串都是回文串
        for (int i = 0; i < n; i++) {
            dp[i][i] = true;
        }
        // 递推开始
        // 先枚举子串长度
        for (int L = 2; L <= n; L++) {
            // 枚举左边界，左边界的上限设置可以宽松一些
            for (int i = 0; i < n; i++) {
                // 由 L 和 i 可以确定右边界，即 j - i + 1 = L 得
                int j = i+L - 1;
                // 如果右边界越界，就可以退出当前循环
                if (j >= n) {
                    break;
                }

                if (s[i] != s[j]) {
                    dp[i][j] = false;
                } else {
                    if (j - i < 3) {
                        dp[i][j] = true;
                    } else {
                        dp[i][j] = dp[i + 1][j - 1];
                    }
                }

                // 只要 dp[i][L] == true 成立，就表示子串 s[i..L] 是回文，此时记录回文长度和起始位置
                if (dp[i][j] && j - i + 1 > maxLen) {
                    maxLen = j - i + 1;
                    begin = i;
                }
            }
        }
        return s.substr(begin, maxLen);
    }
};
```

### 完全背包压缩

#### 完全背包问题：深度解析与矩阵演变

理解了 0/1 背包的“倒序”逻辑后，学习\*\*完全背包（Complete Knapsack）\*\*会非常快，因为它们就像是一枚硬币的正反面。

#### 1. 完全背包的规则

- **0/1 背包**：每个物品只有 1 件。选了就没法再选（要么拿 0 件，要么拿 1 件）。
- **完全背包**：每个物品有**无数件**。只要背包装得下，你可以一直往里塞同一个物品（可以拿 0, 1, 2, … k 件）。

---

#### 2. 二维逻辑的微小差异（决定性的差异）

我们对比一下两者的转移方程：

- **0/1 背包**：  
   `dp[i][j] = max(dp[i-1][j], dp[i-1][j - w[i]] + v[i])`  
   \*注意：选了物品 i 之后，要去\*\*上一行（i-1）\**找剩下容量的最大价值。*
- **完全背包**：  
   `dp[i][j] = max(dp[i-1][j], dp[i][j - w[i]] + v[i])`  
   \*注意：选了物品 i 之后，依然在\*\*这一行（i）\**找剩下容量的最大价值。*

 因为物品有无数件。当你拿了一个物品

#### 4. 一维压缩：为什么要正序？

在一维数组优化时：

- **0/1 背包**为了避开“这一行”的新数据，被迫**倒着走**。
- **完全背包**为了利用“这一行”的新数据，恰恰需要**正着走**！

##### 正序推导过程（例子）：

物品 A：重量 2，价值 10。背包容量 5。  
 数组 `dp` 初始：`[0, 0, 0, 0, 0, 0]`

1. **j = 2**：能放 A。`dp[2] = max(0, dp[2-2] + 10) = 10`。
2. **j = 3**：能放 A。`dp[3] = max(0, dp[3-2] + 10) = 10`。
3. **j = 4**：能放 A。重点来了！
   - 我们要看 `dp[4-2]` 也就是 `dp[2]`。
   - 此时 `dp[2]` 已经在刚才第 1 步更新成了 **10**。
   - 计算：`10 (dp[2]新值) + 10 (物品价值) = 20`。
   - **物理意义**：我们在容量为 2 的基础上又放了一个物品 A，总共放了两个 A。
4. **j = 5**：
   - 看 `dp[5-2]` 也就是 `dp[3]`。
   - 计算：`10 (dp[3]新值) + 10 = 20`。

**这就是正序的神奇之处：** 它允许状态从小容量转移到大容量时，不断累加同一个物品。

#### 5. 总结你的整理笔记

你可以这样记录完全背包的压缩原理：

1. **核心规则**：物品无限选，因此当前状态 `dp[j]` 依赖于**当前行**左侧已经更新过的状态 `dp[j - w_i]`。
2. **压缩机制**：同样使用一维数组覆盖，但由于我们需要利用“这一行”的新结果，所以必须\*\*从左往右（正序）\*\*遍历。
3. **对比结论**：
   - **0/1 背包倒序**：为了拿到**上一行**的旧值（保证物品只用一次）。
   - **完全背包正序**：为了拿到**这一行**的新值（允许物品多次使用）。

#### 6.一个有趣的直观感受：

- **0/1 背包**：你是一个精明的商人，进货单上每个商品只有一件，你必须根据昨天的库存表（上一行）来决定今天要不要拿。为了不搞混，你得从货架最右边开始看，确保参考的每一项都是昨天的旧数据。
- **完全背包**：你是一个批发商，仓库里货多的是。你今天刚把一件货放进背包，发现还能放，就原地转身（参考这一行左边的结果）又拿了一件一模一样的塞了进去。你正着走，就是为了能不断看到自己刚刚放进去的东西，然后决定要不要“再来一个”。

**总结：**

- **0/1 背包**：**倒序**遍历 = **躲避**新值 = **只用一次**。
- **完全背包**：**正序**遍历 = **迎接**新值 = **无限多次**。

为了巩固你的理解，我们把你的这段话转化成标准的算法逻辑表述：

##### 1. 0/1 背包的依赖逻辑（你的理解：全看上一行）

- **不拿**：`dp[i][j] = dp[i-1][j]` —— 直接继承**上一行**同列的值。
- **拿**：`dp[i][j] = dp[i-1][j - w[i]] + v[i]` —— 参考**上一行**左侧的值。
- **核心点**：无论拿不拿，都只看\*\*“上一行”\*\*（即还没开始处理当前物品时的“旧世界”）。
- **所以压缩时**：必须**从后往前**算，防止“上一行”的旧数据被这一行的新数据过早覆盖。

##### 2. 完全背包的依赖逻辑（你的理解：拿的话看这一行）

- **不拿**：`dp[i][j] = dp[i-1][j]` —— 依然是继承**上一行**同列的值（代表一个都不拿）。
- **拿**：`dp[i][j] = dp[i][j - w[i]] + v[i]` —— 参考**这一行**左侧的值。
- **核心点**：如果拿，就看\*\*“这一行”\*\*（即已经处理过当前物品、可能已经放过该物品的状态）。
- **所以压缩时**：必须**从前往后**算，因为我们刚好需要用到这一行左边刚刚更新好的“新数据”。
