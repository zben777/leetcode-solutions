### 一、递归

> 递归算法：  
>  这里帮助大家确定下来递归算法的三个要素。**每次写递归，都按照这三要素来写，可以保证大家写出正确的递归算法**！

1. **确定递归函数的参数和返回值**： 确定哪些参数是递归的过程中需要处理的，那么就在递归函数里加上这个参数， 并且还要明确每次递归的返回值是什么进而确定递归函数的返回类型。
2. **确定终止条件**： 写完了递归算法, 运行的时候，经常会遇到栈溢出的错误，就是没写终止条件或者终止条件写的不对，操作系统也是用一个栈的结构来保存每一层递归的信息，如果递归没有终止，操作系统的内存栈必然就会溢出。
3. **确定单层递归的逻辑**： 确定每一层递归需要处理的信息。在这里也就会重复调用自己来实现递归的过程。

```
前序遍历的例子：来解释递归的过程；
class Solution {
public:
```

### 二、迭代

> 因为三种的迭代有的大不相同，所以代码随想录上面有个是 套路模板；可以仅仅调节几行代码的顺序就可以进行 三种 非递归即迭代的实现；

```
中序遍历代码如下：（详细注释）
class Solution {
public:
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != NULL) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != NULL) {
                st.pop(); // 将该节点弹出，避免重复操作，下面再将右中左节点添加到栈中
                if (node->right) st.push(node->right);  // 添加右节点（空节点不入栈）

                st.push(node);                          // 添加中节点
                st.push(NULL); // 中节点访问过，但是还没有处理，加入空节点做为标记。

                if (node->left) st.push(node->left);    // 添加左节点（空节点不入栈）
            } else { // 只有遇到空节点的时候，才将下一个节点放进结果集
                st.pop();           // 将空节点弹出
                node = st.top();    // 重新取出栈中元素
                st.pop();
                result.push_back(node->val); // 加入到结果集
            }
        }
        return result;
    }
};



迭代法前序遍历代码如下： (注意此时我们和中序遍历相比仅仅改变了两行代码的顺序)
class Solution {
public:
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != NULL) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();

                if (node->left) st.push(node->left);    // 左
                st.push(node);                          // 中
                st.push(NULL);
            } else {
                st.pop();
                node = st.top();
                st.pop();
                result.push_back(node->val);
            }
        }
        return result;
    }
};


后续遍历代码如下： (注意此时我们和中序遍历相比仅仅改变了两行代码的顺序)
class Solution {
public:
    vector<int> postorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != NULL) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != NULL) {
                st.pop();
                st.push(node);                          // 中
                st.push(NULL);

                if (node->right) st.push(node->right);  // 右
                if (node->left) st.push(node->left);    // 左

            } else {
                st.pop();
                node = st.top();
                st.pop();
                result.push_back(node->val);
            }
        }
        return result;
    }
};
```

### 三、层序

> 利用queue 进行的实现

```
// 模板如下：
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        queue<TreeNode*> que;
        if (root != NULL) que.push(root);
        vector<vector<int>> result;
        while (!que.empty()) {
            int size = que.size();
            vector<int> vec;
            // 这里一定要使用固定大小size，不要使用que.size()，因为que.size是不断变化的
            for (int i = 0; i < size; i++) {
                TreeNode* node = que.front();
                que.pop();
                vec.push_back(node->val);
                if (node->left) que.push(node->left);
                if (node->right) que.push(node->right);
            }
            result.push_back(vec);
        }
        return result;
    }
};
```

```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}

class Solution {
public:
    void traversal(TreeNode* node, vector<int>& res){
        if(node == nullptr)return;
        traversal(node->left, res);
        res.push_back(node->val);
        traversal(node->right, res);

        traversal(root, res);
```

```
class Solution {
public:

class Solution {
```

### 题目来源：LeetCode-101-对称二叉树

题目描述：  
 ![在这里插入图片描述](images/129705622_3.png)

题目思路：

> 对于二叉树是否对称，要比较的是根节点的**左子树与右子树是不是相互翻转的**，理解这一点就知道了其实我们要比较的是两个树（这两个树是根节点的左右子树），所以在递归遍历的过程中，也是要同时遍历两棵树。  
>  正是因为要遍历两棵树而且要比较内侧和外侧节点，所以准确的来说是一个树的遍历顺序是左右中，一个树的遍历顺序是右左中。

题目代码：

```
class Solution {
public:
    bool isSameTree(TreeNode* left_tree, TreeNode* right_tree){
        if(left_tree == nullptr || right_tree == nullptr){
            if(left_tree == right_tree)return true;
            return false;
        }
        if(left_tree->val != right_tree->val)return false;

        // 真正的递归是 左边的树的左边的孩子 和 右边的树的右边的孩子比;
        //  左边的树的右边的孩子 和 右边的树的左边的孩子比;
        bool child_left = isSameTree(left_tree->left, right_tree->right);
        if(child_left == false)return false;

        bool child_right = isSameTree(left_tree->right, right_tree->left);
        if(child_right == false)return false;
        return true;
    }
    bool isSymmetric(TreeNode* root) {
        if (root == nullptr) return true;

        // 如果考虑递归的话
        // 就是root的左边和右边。
        return isSameTree(root->left, root->right);
    }
};




首先是递归的三部曲：
1：确定递归函数的参数和返回值
因为我们要比较的是根节点的两个子树是否是相互翻转的，进而判断这个树是不是对称树，所以要比较的是两个树，参数自然也是左子树节点和右子树节点。

class Solution {

        }
```

```
        return root;
    }
```

```
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int res = 0;  // 记录答案总数
    void dfs(TreeNode* root, long long sum, const int& targetSum){
        if(root == nullptr)return;
        sum += root->val;
        if(sum == targetSum)res++;
        dfs(root->left, sum, targetSum);
        dfs(root->right, sum, targetSum);
        sum -= root->val;
    }

    int pathSum(TreeNode* root, int targetSum) {
        res = 0;
        if(root == nullptr)return 0;

        // 也就是说遍历所有的节点 dfs
        queue<TreeNode*>qt;
        qt.push(root);
        while(!qt.empty()){
            int sz = qt.size();
            for(int i = 0; i < sz; i++){
                auto ff = qt.front();
                dfs(ff, 0, targetSum);
                qt.pop();
                if(ff->left)qt.push(ff->left);
                if(ff->right)qt.push(ff->right);
            }
        }
        return res;
    }
};
```

### 题目来源：LeetCode-124. 二叉树中的最大路径和

题目描述：  
 ![在这里插入图片描述](images/129705622_15.png)

题目思路：

> 完全沿用统一的递归模板（左→右→根处理），  
>  递归算左子树最大贡献值  
>  递归算右子树最大贡献值  
>  更新全局最大路径和（左 + 根 + 右）  
>  返回根 + 最大单边给上层使用  
>  全程就是最熟悉的：先左、再右、最后处理根

题目代码：

```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int maxSum = -INT_MAX;  
    // 递归：返回当前节点能给父节点的最大贡献
    int dfs(TreeNode* root) {
        if (root == nullptr) return 0;

        // local 
        int local_sum = root->val;
        int left_sum = max(dfs(root->left), 0);
        int right_sum = max(dfs(root->right), 0);
        local_sum += left_sum;
        local_sum += right_sum;
        maxSum = max(maxSum, local_sum);

        return root->val + max(left_sum, right_sum);
    }

    int maxPathSum(TreeNode* root) {
        dfs(root);
        return maxSum;
    }
};












/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int maxSum = INT_MIN;  // 全局记录答案

    // 递归：返回当前节点能给父节点的最大贡献
    int dfs(TreeNode* root) {
        if (!root) return 0;

        // 左子树贡献（负数就不取，取0）
        int left = max(dfs(root->left), 0);
        // 右子树贡献
        int right = max(dfs(root->right), 0);

        // 更新全局最大路径：左 + 根 + 右
        maxSum = max(maxSum, left + root->val + right);

        // 返回给上层：只能选一条边走（根 + 最大单边）
        return root->val + max(left, right);
    }

    int maxPathSum(TreeNode* root) {
        dfs(root);
        return maxSum;
    }
};
```

>  或者说可以层序遍历，然后保留最底层的然后就行了

> 使用广度优先搜索遍历每一层的节点。在遍历一个节点时，需要先把它的非空右子节点放入队列，然后再把它的非空左子节点放入队列，这样才能保证从右到左遍历每一层的节点。广度优先搜索所遍历的最后一个节点的值就是最底层最左边节点的值。

```
 *     TreeNode *right;

 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };

            }
        }

    }
```

```
public:
    void inorderTraversal(TreeNode* now, vector<int> &ans, const int k){
        if(now == nullptr)return;
        inorderTraversal(now->left, ans, k);
        ans.push_back(now->val);
        if(ans.size() == k)return;
        inorderTraversal(now->right, ans, k);

    int kthSmallest(TreeNode* root, int k) {
```

```
class Solution {
public:
    Node* inorderSuccessor(Node* node) {
        if(!node) return nullptr;
        
        // 如果node有右子树，则后继节点是右子树最左节点
        if(node->right){
            node = node->right;
            while(node->left)
                node = node->left;
            return node;
        }else{
            // 如果node没有右子树
            // 1：如果node是父节点的左孩子，则后继节点是父节点
            // 2: 如果node是父节点右孩子，则不断向上找到某个节点p，是其父节点的左孩子，则该父节点为后继
            Node* parent = node->parent;
            while(parent && parent->left!=node){
                node = parent;
                parent = node->parent;
            }
            return parent;
        }
    }
```

```
            return 1;
```

```
class Trie {
private:
    vector<Trie*> children;
    bool isEnd;

    Trie* searchPrefix(string prefix) {
        Trie* node = this;
        for (char ch : prefix) {
            ch -= 'a';
            if (node->children[ch] == nullptr) {
                return nullptr;
            }
            node = node->children[ch];
        }
        return node;
    }

public:
    Trie() : children(26), isEnd(false) {}

    void insert(string word) {
        Trie* node = this;
        for (char ch : word) {
            ch -= 'a';
            if (node->children[ch] == nullptr) {
                node->children[ch] = new Trie();
            }
            node = node->children[ch];
        }
        node->isEnd = true;
    }

    bool search(string word) {
        Trie* node = this->searchPrefix(word);
        return node != nullptr && node->isEnd;
    }

    bool startsWith(string prefix) {
        return this->searchPrefix(prefix) != nullptr;
    }
};
```
