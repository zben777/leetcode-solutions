```cpp
ListNode* reverse(ListNode* head) {
    ListNode* pre = nullptr;
    while (head) {
        auto next = head->next;
        head->next = pre;
        pre = head;
        head = next;
    }
    return pre;
}

ListNode* reverseList(ListNode* head) {
    // 新建一个链表头，完全不碰原来的链表
    ListNode* pre = nullptr;
    ListNode* cur = head;  // 只遍历，不修改原链表
    
    while (cur) {
        // 【关键】每次创建新节点，不改动原来的节点！
        ListNode* newNode = new ListNode(cur->val);
        // 头插法
        newNode->next = pre;
        pre = newNode;

        cur = cur->next;
    }
    return pre;
}
```

##### ③ 环形链表（141判环 +142找入环点，附数学推导）

##### ③数学结论（图中公式文字整理）

![在这里插入图片描述](images/129759679_1.png)

- 原始到入环的为x， 入环到相遇为y 相遇到入环为z 则 slow = x + y fast = x + y + z + y ;且 fast = 2 slow 则得到 z = x
- 设：**头→入环口 = X 步**，环总长 = `Y + Z`；相遇时慢指针总路程`X + Y`，快指针总路程`X+ Y + Z + Y`
- `快慢速度2倍关系`：`2X + 2Y = X + 2Y + Z`（快比慢多走k整圈）
- 得到路程关系：`X = Z`
- **关键结论：从相遇点再走Z步恰好抵达入环口**

> 操作：一个指针从**链表头**、一个指针从**相遇点**同步一步一步走，首次相遇位置 = 环入口

```cpp
//141 判断是否有环
bool hasCycle(ListNode *head) {
    ListNode* slow=head,*fast=head;
    while(fast&&fast->next){
        slow=slow->next;
        fast=fast->next->next;
        if(slow==fast) return true;
    }
    return false;
}

//142 返回环入口节点
ListNode *detectCycle(ListNode *head) {
    ListNode* slow=head,*fast=head;
    while(fast&&fast->next){
        slow=slow->next;
        fast=fast->next->next;
        if(slow==fast){
            //相遇，头指针和慢指针同速出发
            ListNode* p=head;
            while(p!=slow){
                p=p->next;
                slow=slow->next;
            }
            return p;
        }
    }
    return nullptr;
}
```

### 题目来源：LeetCode-160. 相交链表

题目描述：  
 ![在这里插入图片描述](images/129759679_3.png)

题目思路：

> 先是暴力进行的  
>  如果符合O(m + n)的话，还必须是双指针
>
> 1. 核心痛点:  
>     两个链表可能长度不同，直接遍历无法对齐相交节点（比如 A 链表长 5，B 链表长 3，相交节点在 A 的第 3 位、B 的第 1 位）。
> 2. 解决思路  
>     让两个指针交叉遍历两个链表：  
>     指针 A 先遍历完链表 A，再从链表 B 的头节点开始遍历；  
>     指针 B 先遍历完链表 B，再从链表 A 的头节点开始遍历；  
>     最终两个指针会在相交节点相遇（若相交），或同时走到NULL（若不相交）。
> 3. 数学证明（为什么能相遇？）

题目代码：

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        //这个记住 当 A跑了全程+B的第一段
        // 当B跑了全程+A的第一段； 上面的和下面的是一样的;
        ListNode* cura = headA;
        ListNode* curb = headB;
        if(cura == nullptr || curb == nullptr)return nullptr;

        while(cura != curb){
            // a跑
            if(cura == nullptr){
                cura = headB;
            }else{
                cura = cura->next;
            }

            // b跑
            if(curb == nullptr){
                curb = headA;
            }else{
                curb = curb->next;
            }
        }
        return cura;
    }
};
```

        // 指针移动

        // 相交的节点是。地址相等，我靠
        std::vector<ListNode*> node_ptrs;
        while (headA != nullptr) {
            node_ptrs.push_back(headA); // 存指针，不是存val
            headA = headA->next;
        }

        // 遍历链表B，检查当前节点的指针是否在vector中
        while (headB != nullptr) {
            auto it = std::find(node_ptrs.begin(), node_ptrs.end(), headB);
            if (it != node_ptrs.end()) {
                // 找到相同指针，说明是相交节点
                return headB;
            }
            headB = headB->next;
        }
        return nullptr;

    }
};

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        // 边界处理：空链表直接返回NULL
        if (headA == nullptr || headB == nullptr) {
            return nullptr;
        }

        // 初始化两个指针，分别指向两个链表的头节点
        ListNode* pA = headA;
        ListNode* pB = headB;

        // 核心循环：直到两个指针地址相等
        // （相等时要么是相交节点，要么都是NULL）
        while (pA != pB) {
            // 指针A：没走到头就后移，走到头就切换到B的头
            pA = (pA == nullptr) ? headB : pA->next;
            // 指针B：没走到头就后移，走到头就切换到A的头
            pB = (pB == nullptr) ? headA : pB->next;
        }

        // 循环结束时，pA和pB要么是相交节点，要么是NULL
        return pA;
    }
};
```

题目描述：

题目思路：



题目代码：

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        // 虚拟头节点（哨兵节点），避免空指针处理，简化逻辑
        ListNode* dummy = new ListNode();
        // 移动指针，用于构建新链表
        ListNode* cur = dummy;

        while(list1 && list2){
            if(list1->val <= list2->val){
                ListNode* newNode = new ListNode(list1->val);
                cur->next = newNode;
                cur = cur->next;
                list1 = list1->next;
            }else{
                ListNode* newNode = new ListNode(list2->val);
                cur->next = newNode;
                cur = cur->next;
                list2 = list2->next;
            }
        }
        if(list1)cur->next = list1;
        if(list2)cur->next = list2;
        return dummy->next;
    }
};
```

### 题目来源：LeetCode-2. 两数相加

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        // 虚拟头节点：简化链表拼接的边界处理（避免空指针）
        ListNode* dummy = new ListNode(0);
        // 移动指针：用于拼接新节点
        ListNode* cur = dummy;

        int carry = 0;
        while(l1 || l2 || carry){
            int sum = 0;
            if(l1){
                sum += l1->val;
                l1 = l1->next;
            }
            if(l2){
                sum += l2->val;
                l2 = l2->next;
            }
            sum += carry;

            if(sum >= 10){
                carry = sum / 10;
                sum = sum % 10;
                ListNode* newNode = new ListNode(sum);
                cur->next = newNode;
                cur = cur->next;
            }else{
                carry = 0;
                ListNode* newNode = new ListNode(sum);
                cur->next = newNode;
                cur = cur->next;
            }
        }
        return dummy->next;
    }
};
```

### 题目来源：LeetCode-138. 随机链表的复制

题目描述：  
 ![在这里插入图片描述](images/129759679_6.png)

题目思路：



题目代码：

```cpp
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        if(head == nullptr)return head;
        Node* dummy = new Node(0);
        dummy->next = head;
        Node* cur = head;
        // 对于每一个都节点 都先创建一个副本；跳的时候跳2hops
        for(cur = head; cur != nullptr; cur = cur->next->next){
            Node* newNode = new Node(cur->val);
            newNode->next = cur->next;
            cur->next = newNode;
        }
        // 现在上面的val 和 next好了
        for(cur = head; cur != nullptr; cur = cur->next->next){
            Node* newNode = cur->next;
            if(cur->random == nullptr){
                newNode->random = nullptr;
            }else{ //记住这里
                newNode->random = cur->random->next;
            }
        }
        // 现在 random 好了 
        // 下一步进行清除之前的
        dummy = head->next;
        for(cur = head; cur != nullptr; cur = cur->next){
            Node* newNode = cur->next;
            cur->next = cur->next->next;
            if(newNode->next == nullptr){ // 后面没节点了
                newNode->next = nullptr;
            }else{
                newNode->next = newNode->next->next;
            }
        }
        return dummy;
    }
};
```

```cpp
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        // 边界：空链表直接返回
        if (head == nullptr) return nullptr;

        // 步骤1：插桩 - 在原节点后插入复制节点
        Node* p = head;
        while (p != nullptr) {
            // 创建复制节点，val和原节点相同，next指向原节点的next
            Node* copy_node = new Node(p->val);
            copy_node->next = p->next;
            // 原节点的next指向复制节点
            p->next = copy_node;
            // 移动到下一个原节点
            p = copy_node->next;
        }

        // 步骤2：复制random指针
        p = head;
        while (p != nullptr) {
            Node* copy_node = p->next; // 复制节点是原节点的next
            // 若原节点的random不为空，则复制节点的random指向原random的next（即random的复制节点）
            if (p->random != nullptr) {
                copy_node->random = p->random->next;
            }
            // 移动到下一个原节点（跳过复制节点）
            p = copy_node->next;
        }

        // 步骤3：拆分链表 - 分离原链表和复制链表
        p = head;
        Node* new_head = head->next; // 复制链表的头节点
        Node* copy_p = new_head;
        while (p != nullptr) {
            // 恢复原节点的next（指向原下一个节点）
            p->next = copy_p->next;
            p = p->next; // 移动原节点指针
            
            // 若还有复制节点，更新复制节点的next
            if (p != nullptr) {
                copy_p->next = p->next;
                copy_p = copy_p->next;
            }
        }

        return new_head;
    }
};
```

### 题目来源：LeetCode-146. LRU 缓存

题目描述：  
 ![在这里插入图片描述](images/129759679_7.png)

题目思路：

>  1-自己设计Node，然后整双向链表  
>  2-直接使用C++自带的list标准库

题目代码：

```cpp
struct Node {
    int key;
    int value;
    Node* prev;
    Node* next;

    Node(int k = 0, int v = 0) : key(k), value(v) {}
};

class LRUCache {
private:
    int capacity;
    Node* dummy; // 哨兵节点
    unordered_map<int, Node*> key_to_node;

    // 删除一个节点（抽出一本书）
    void remove(Node* x) {
        x->prev->next = x->next;
        x->next->prev = x->prev;
    }

    // 在链表头添加一个节点（把一本书放到最上面）
    void push_front(Node* x) {
        x->prev = dummy;
        x->next = dummy->next;
        x->prev->next = x;
        x->next->prev = x;
    }

    // 获取 key 对应的节点，同时把该节点移到链表头部
    Node* get_node(int key) {
        auto it = key_to_node.find(key);
        if (it == key_to_node.end()) { // 没有这本书
            return nullptr;
        }
        Node* node = it->second; // 有这本书
        remove(node); // 把这本书抽出来
        push_front(node); // 放到最上面
        return node;
    }

public:
    LRUCache(int capacity) : capacity(capacity), dummy(new Node()) {
        dummy->prev = dummy;
        dummy->next = dummy;
    }

    int get(int key) {
        Node* node = get_node(key); // get_node 会把对应节点移到链表头部
        return node ? node->value : -1;
    }

    void put(int key, int value) {
        Node* node = get_node(key); // get_node 会把对应节点移到链表头部
        if (node) { // 有这本书
            node->value = value; // 更新 value
            return;
        }
        key_to_node[key] = node = new Node(key, value); // 新书
        push_front(node); // 放到最上面
        if (key_to_node.size() > capacity) { // 书太多了
            Node* back_node = dummy->prev;
            key_to_node.erase(back_node->key);

class LRUCache {
private:
    int capacity;
    list<pair<int, int>> cache_list; // pair 里面保存的是 key 和 value
    unordered_map<int, list<pair<int, int>>::iterator> key_to_iter; // key -> 链表节点迭代器

public:
    LRUCache(int capacity) : capacity(capacity) {}

    int get(int key) {
        auto umap_iter = key_to_iter.find(key);
        if (umap_iter == key_to_iter.end()) { // 没有这本书
            return -1;
        }
        auto list_iter = umap_iter->second; // 有这本书
        // 把这本书（list_iter）从书堆（cache_list）中抽出来，放到最上面（cache_list.begin()）
        cache_list.splice(cache_list.begin(), cache_list, list_iter);
        return list_iter->second; // 返回这本书的 value
    }

    void put(int key, int value) {
        auto umap_iter = key_to_iter.find(key);
        if (umap_iter != key_to_iter.end()) { // 有这本书
            auto list_iter = umap_iter->second;
            list_iter->second = value; // 更新 value
            // 把这本书（list_iter）从书堆（cache_list）中抽出来，放到最上面（cache_list.begin()）
            cache_list.splice(cache_list.begin(), cache_list, list_iter);
            return;
        }
        // 新书，放到最上面（emplace_front）
        cache_list.emplace_front(key, value);
        key_to_iter[key] = cache_list.begin();
        // 书太多了
        if (key_to_iter.size() > capacity) {
            // 去掉最后一本书
            key_to_iter.erase(cache_list.back().first);
            cache_list.pop_back();
        }
    }
};

    ListNode* sortList(ListNode* head) {
        // 如果链表为空或者只有一个节点，无需排序
        if (head == nullptr || head->next == nullptr) {
            return head;
        }
        // 找到中间节点 head2，并断开 head2 与其前一个节点的连接
        // 比如 head=[4,2,1,3]，那么 middleNode 调用结束后 head=[4,2] head2=[1,3]
        ListNode* head2 = middleNode(head);
        // 分治
        head = sortList(head);
        head2 = sortList(head2);
        // 合并
        return mergeTwoLists(head, head2);
    }
};
```

class Solution {
    // 21. 合并两个有序链表
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        ListNode dummy{}; // 用哨兵节点简化代码逻辑
        auto cur = &dummy; // cur 指向新链表的末尾
        while (list1 && list2) {
            if (list1->val < list2->val) {
                cur->next = list1; // 把 list1 加到新链表中
                list1 = list1->next;
            } else { // 注：相等的情况加哪个节点都是可以的
                cur->next = list2; // 把 list2 加到新链表中
                list2 = list2->next;
            }
            cur = cur->next;
        }
        cur->next = list1 ? list1 : list2; // 拼接剩余链表
        return dummy.next;
    }

public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        int m = lists.size();
        if(m == 0)return nullptr;

        // 像是规约了
        for(int step = 1; step < m; step = step * 2){
            for(int j = 0; j + step < m; j = j + step * 2){
                lists[j] = mergeTwoLists(lists[j], lists[j + step]);
            }
        }
        return lists[0];
    }
};

        }
        cur->next = list1 ? list1 : list2; // 拼接剩余链表
        return dummy.next;
    }

    // 合并从 lists[i] 到 lists[j-1] 的链表
    ListNode* mergeKLists(vector<ListNode*>& lists, int i, int j) {
        int m = j - i;
        if (m == 0) {
            return nullptr; // 注意输入的 lists 可能是空的
        }
        if (m == 1) {
            return lists[i]; // 无需合并，直接返回
        }
        auto left = mergeKLists(lists, i, i + m / 2); // 合并左半部分
        auto right = mergeKLists(lists, i + m / 2, j); // 合并右半部分
        return mergeTwoLists(left, right); // 最后把左半和右半合并
    }

public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        return mergeKLists(lists, 0, lists.size());
    }
};
