### 十:二分查找

```cpp
class Solution {
public:
    int search(vector<int>& nums, int target) {

        int l=0,r=nums.size()-1;

        while(l<=r){
            int mid = (l+r)/2;
            if(nums[mid]==target){
                return mid;
            }else if(nums[mid]<target){
                l=l+1;
            }else if(nums[mid]>target){
                r=r-1;
            }
        }

        return -1;

    }
};
```

        cout<<"未找到字符串"<<endl;

##### 5：字符串的比较

> 逐个按照字符 的ASCII码值进行比较  
>  ==返回0； > 返回1；；<返回-1；；

```cpp
#include <bits/stdc++.h>
using namespace std;

int main()
{
    string str1="xello";
    string str2="hello";
    int res=str1.compare(str2);

    if(res==0){
        cout<<"两个字符串相等"<<endl;
    }
    else if(res>0){
        cout<<"大于"<<endl;
    }
    else if(res<0){
        cout<<"小于"<<endl;
    }
    return 0;
}
```
